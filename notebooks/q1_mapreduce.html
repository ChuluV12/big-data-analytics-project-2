Question 1 - Keyword location search with MapReduce (R, Hadoop Streaming)

The mapper and reducer are plain R scripts. Hadoop Streaming runs them as the map and reduce tasks; here the same two scripts run in a local pipeline (map, sort by key, reduce), which is the data flow Hadoop performs on a cluster.

Mapper

#!/usr/bin/env Rscript
# mapper.R
# Reads one input split from stdin and emits one record per keyword match:
#     <file name> TAB <row>,<column>
# The keyword is passed as the first command-line argument, so nothing is hardcoded.

args <- commandArgs(trailingOnly = TRUE)
if (length(args) < 1) stop("Usage: Rscript mapper.R <keyword>")
keyword <- args[1]

# Hadoop Streaming exposes the current input file as this environment variable.
file_name <- basename(Sys.getenv("mapreduce_map_input_file", unset = "stdin"))

# Whole-word, case-insensitive match. \\Q...\\E treats the keyword as plain text.
pattern <- paste0("\\b\\Q", keyword, "\\E\\b")

con   <- file("stdin", encoding = "UTF-8")
lines <- readLines(con, warn = FALSE)
close(con)

for (row in seq_along(lines)) {
  hits <- gregexpr(pattern, lines[row], ignore.case = TRUE, perl = TRUE)[[1]]
  if (hits[1] != -1) {
    cat(sprintf("%s\t%d,%d\n", file_name, row, as.integer(hits)), sep = "")
  }
}

Reducer

#!/usr/bin/env Rscript
# reducer.R
# Input (sorted by key):  <file name> TAB <row>,<column>
# Output: <file name> TAB <number of matches> TAB (row,col) (row,col) ...

con   <- file("stdin", encoding = "UTF-8")
lines <- readLines(con, warn = FALSE)
close(con)
lines <- lines[nzchar(lines)]
if (length(lines) == 0) quit(save = "no")

parts <- strsplit(lines, "\t", fixed = TRUE)
key   <- vapply(parts, `[`, character(1), 1)
value <- vapply(parts, `[`, character(1), 2)

for (k in sort(unique(key))) {
  loc <- do.call(rbind, strsplit(value[key == k], ",", fixed = TRUE))
  row <- as.integer(loc[, 1])
  col <- as.integer(loc[, 2])
  ord <- order(row, col)
  cat(sprintf("%s\t%d\t%s\n", k, length(ord),
              paste0("(", row[ord], ",", col[ord], ")", collapse = " ")))
}
