Question 2 - LDA topic modelling of the news files

The script q2_lda.R builds the data frame, cleans the text, fits the model and saves the figures. Running it here prints each step.

source("q2_lda.R", echo = TRUE, max.deparse.length = Inf)

> files <- list.files("data/news", pattern = "\\.txt$", 
+     full.names = TRUE)

> files <- files[order(as.integer(gsub("\\D", "", basename(files))))]

> news <- data.frame(file_name = basename(files), content = unname(vapply(files, 
+     function(f) paste(readLines(f, warn = FALSE, encoding = "UTF-8"), 
+         collapse = " "), character(1))), stringsAsFactors = FALSE)

> days <- c("monday", "tuesday", "wednesday", "thursday", 
+     "friday", "saturday", "sunday")

> months <- tolower(month.name)

> extra_stop <- c("said", "says", "also", "will", "must", 
+     "can", "may", "one", "two", "new", "per", "cent", "according", 
+     "including")

> clean_text <- function(x) {
+     x <- gsub("[\\x{2018}\\x{2019}]", "'", x, perl = TRUE)
+     x <- gsub("[\\x{201C}\\x{201D}]", "\"", x, perl = TRUE)
+     x <- gsub("[\\x{2010}-\\x{2015}]", " ", x, perl = TRUE)
+     x <- tolower(x)
+     x <- gsub("http\\S+|www\\.\\S+|\\S+@\\S+", " ", x)
+     x <- gsub("'s\\b", "", x)
+     x <- removeWords(x, c(stopwords("en"), days, months, extra_stop))
+     x <- gsub("[^a-z ]", " ", x)
+     x <- gsub("\\b[a-z]{1,2}\\b", " ", x)
+     x <- stripWhitespace(x)
+     trimws(x)
+ }

> news$clean <- clean_text(news$content)

> dtm_all <- DocumentTermMatrix(VCorpus(VectorSource(news$clean)))

> doc_freq <- slam::col_sums(dtm_all > 0)

> dtm <- dtm_all[, doc_freq >= 2 & doc_freq <= 0.6 * 
+     nrow(dtm_all)]

> rownames(dtm) <- news$file_name

> dtm <- dtm[slam::row_sums(dtm) > 0, ]

> cat("Documents:", nrow(dtm), " Terms kept:", ncol(dtm),

Documents: 10  Terms kept: 188
